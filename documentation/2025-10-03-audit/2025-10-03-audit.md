### 🔴 CRÍTICO #1: Memory Leak en Retry Logic

**Archivo**: `lib/presentation/screens/movies/movie_screen.dart`  
**Líneas**: 23-42  
**Severidad**: 🔴 CRÍTICO  
**Impacto**: Memory leaks, crashes en dispositivos con poca RAM

**Problema**:

```dart
// ❌ PROBLEMA ACTUAL
Future<void> _loadMovieWithRetry() async {
  await Future(() async {
    try {
      await ref.read(movieInfoProvider.notifier).loadMovie(widget.movieId);
    } catch (e) {
      if (_retryCount < _maxRetries && mounted) {
        _retryCount++;
        await Future.delayed(Duration(seconds: _retryCount));
        if (mounted) {
          _loadMovieWithRetry(); // ← Recursión sin cleanup
        }
      }
    }
  });
}
```

**Problemas identificados**:

1. No cancela retries al hacer dispose
2. Puede causar memory leaks si el widget se desmonta durante retry
3. No informa al usuario del error final
4. Recursión sin control de ciclo de vida

**Solución propuesta**:

```dart
// ✅ SOLUCIÓN
class MovieScreenState extends ConsumerStatefulWidget {
  int _retryCount = 0;
  bool _disposed = false; // ← AGREGAR

  @override
  void dispose() {
    _disposed = true; // ← AGREGAR
    super.dispose();
  }

  Future<void> _loadMovieWithRetry() async {
    if (_disposed) return; // ← AGREGAR

    try {
      await ref.read(movieInfoProvider.notifier).loadMovie(widget.movieId);
    } catch (e) {
      if (_retryCount < _maxRetries && !_disposed) {
        _retryCount++;
        await Future.delayed(Duration(seconds: _retryCount));
        if (!_disposed) {
          _loadMovieWithRetry();
        }
      } else if (!_disposed && mounted) {
        // Mostrar error al usuario
        _showErrorSnackbar('No se pudo cargar la película');
      }
    }
  }

  void _showErrorSnackbar(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        action: SnackBarAction(
          label: 'Reintentar',
          onPressed: () {
            _retryCount = 0;
            _loadMovieWithRetry();
          },
        ),
      ),
    );
  }
}
```

**Estimación**: 2 horas  
**Prioridad**: 🔴 URGENTE  
**Testing requerido**: Unit tests + Manual testing

```dart
// ✅ IMPLEMENTACIÓN REAL con JustWatch API
class StreamingServiceManager {
  final Dio _dio;
  final String _justWatchApiKey;

  Future<List<StreamingProvider>> getAvailability(
    int movieId, {
    String country = 'US',
  }) async {
    try {
      final response = await _dio.get(
        'https://apis.justwatch.com/content/titles/movie/$movieId/locale/$country',
        options: Options(headers: {'X-API-Key': _justWatchApiKey}),
      );

      return _parseStreamingProviders(response.data);
    } catch (e) {
      LoggerFactory.error('Error fetching streaming availability: $e');
      return [];
    }
  }

  List<StreamingProvider> _parseStreamingProviders(dynamic data) {
    // Parse JustWatch response
    final offers = data['offers'] as List? ?? [];
    return offers.map((offer) => StreamingProvider.fromJson(offer)).toList();
  }
}
```

```dart
// lib/core/network/optimized_image_cache.dart
class OptimizedImageCache {
  static final Map<String, Uint8List> _memoryCache = {};
  static const int maxCacheSize = 50 * 1024 * 1024; // 50MB
  static int _currentCacheSize = 0;

  static Future<Uint8List?> getCachedImage(String url) async {
    // 1. Check memory cache
    if (_memoryCache.containsKey(url)) {
      return _memoryCache[url];
    }

    // 2. Check disk cache
    final diskCache = await _getDiskCache(url);
    if (diskCache != null) {
      _addToMemoryCache(url, diskCache);
      return diskCache;
    }

    // 3. Download and cache
    final bytes = await _downloadImage(url);
    if (bytes != null) {
      await _saveToDiskCache(url, bytes);
      _addToMemoryCache(url, bytes);
    }

    return bytes;
  }

  static void _addToMemoryCache(String url, Uint8List bytes) {
    if (_currentCacheSize + bytes.length > maxCacheSize) {
      _evictOldestEntries(bytes.length);
    }
    _memoryCache[url] = bytes;
    _currentCacheSize += bytes.length;
  }

  static void _evictOldestEntries(int requiredSpace) {
    // LRU eviction
  }

  static Future<void> clearCache() async {
    _memoryCache.clear();
    _currentCacheSize = 0;
    // Clear disk cache
  }
}
```

```dart
class InfiniteMovieList extends StatefulWidget {
  @override
  State<InfiniteMovieList> createState() => _InfiniteMovieListState();

class _InfiniteMovieListState extends State<InfiniteMovieList> {
  final ScrollController _scrollController = ScrollController();
  int _currentPage = 1;
  bool _isLoadingMore = false;

  @override
  void initState() {
    super.initState();
    _scrollController.addListener(_onScroll);
  }

  void _onScroll() {
    if (_scrollController.position.pixels >=
        _scrollController.position.maxScrollExtent * 0.8) {
      _loadMore();
    }
  }

  Future<void> _loadMore() async {
    if (_isLoadingMore) return;

    setState(() => _isLoadingMore = true);

    await ref.read(moviesProvider.notifier).loadPage(_currentPage + 1);

    setState(() {
      _currentPage++;
      _isLoadingMore = false;
    });
  }

  @override
  Widget build(BuildContext context) {
    return ListView.builder(
      controller: _scrollController,
      itemBuilder: (context, index) {
        if (index == items.length) {
          return _LoadingIndicator();
        }
```
